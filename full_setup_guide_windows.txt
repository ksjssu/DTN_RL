# 0) 작업 폴더 이동
cd C:\Users\Public\workspace\git\DTN_RL

# 1) 가상환경 생성 및 필수 라이브러리 설치 (한 번만 필요)
python3 -m venv .venv
.\.venv\Scripts\pip install --index-url https://download.pytorch.org/whl/cpu torch==2.2.1+cpu
.\.venv\Scripts\pip install "numpy<2" matplotlib

# 2) DRL 서버 실행 (배치512, 미니128, 포트5060)
cmd /c "set PPO_BATCH_SIZE=512 && set PPO_MINIBATCH=128 && set DRL_PORT=5060 && .venv\Scripts\python.exe toolkit\drl_server.py > drl_server_run.log 2>&1"

# 3) DRL 서버 PID 확인 (PowerShell에서 실행)
Get-CimInstance Win32_Process | Where-Object { $_.CommandLine -like '*toolkit/drl_server.py*' } | Select Name,ProcessId,CommandLine

# 4) 시뮬레이터 실행 (예: 20회)
java -Xmx2G -cp "target;lib/*" core.DTNSim -b 20 scenarios/dynamic/drl_train/drl_train_buf60_100k.txt

# 5) 실행 후 DRL 서버 종료 (위에서 확인한 PID로 교체)
taskkill /PID <DRL_PID> /F
