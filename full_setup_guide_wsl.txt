# 0) WSL(Ubuntu)에서 작업 디렉터리 이동
cd /mnt/c/Users/Public/workspace/git/DTN_RL

# 1) 가상환경 생성 및 필수 라이브러리 설치 (한 번만 필요)
python3 -m venv .venv
.venv/bin/pip install --index-url https://download.pytorch.org/whl/cpu torch==2.2.1+cpu
.venv/bin/pip install "numpy<2" matplotlib

# 2) DRL 서버 실행 (배치512, 미니128, 포트5060)
PPO_BATCH_SIZE=512 PPO_MINIBATCH=128 DRL_PORT=5060 .venv/bin/python3 toolkit/drl_server.py > drl_server_run.log 2>&1 &

# 3) DRL 서버 PID 확인
pgrep -f "toolkit/drl_server.py"

# 4) 시뮬레이터 실행 (예: 20회)
java -Xmx2G -cp "target:lib/*" core.DTNSim -b 20 scenarios/dynamic/drl_train/drl_train_buf60_100k.txt

# 5) 실행 후 DRL 서버 종료
kill <DRL_PID>
